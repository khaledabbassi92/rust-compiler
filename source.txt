// Helper: Measure string length
proc string_length(str: text) -> int {
    slot len: int = 0;
    slot ptr: *int = str;
    during ((*(ptr + len) % 256) != 0) {
        len = len + 1;
    }
    yield len;
}

// Helper: Extract byte at character index
proc char_at(str: text, index: int) -> int {
    slot ptr: *int = str;
    yield *(ptr + index) % 256;
}

// Helper: Check if character is vowel ('a'=97, 'e'=101, 'i'=105, 'o'=111, 'u'=117)
proc is_vowel(ch: int) -> flag {
    when ((ch == 97) || (ch == 101) || (ch == 105) || (ch == 111) || (ch == 117)) {
        yield yes;
    }
    yield no;
}

// Helper: Find keyword index in an array table (-1 if not found)
proc lookup_keyword(table: *text, table_size: int, query: text) -> int {
    slot q_len: int = string_length(query);
    slot i: int = 0;

    during (i < table_size) {
        slot entry: text = *(table + (i * 8));
        slot e_len: int = string_length(entry);

        when (q_len == e_len) {
            slot match: flag = yes;
            slot c_idx: int = 0;

            during (c_idx < q_len) {
                when (char_at(entry, c_idx) != char_at(query, c_idx)) {
                    match = no;
                }
                c_idx = c_idx + 1;
            }

            when (match) {
                yield i; // Found matching keyword index!
            }
        }

        i = i + 1;
    }

    yield -1; // Not found
}
